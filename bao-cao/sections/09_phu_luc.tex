\subsection*{Cấu trúc kho mã nguồn}

\subsection*{Tái lập kết quả}

\begin{enumerate}[label=\arabic*., leftmargin=1.2cm]
  \item Tải mã nguồn: \texttt{git clone \LinkGitHub}
  \item Cài Python 3.12 và các thư viện: \texttt{pip install pandas numpy scipy matplotlib}
  \item Tính lại bảng và hình của báo cáo: \texttt{python3 analysis/report\_analysis.py}
  \item Tính lại dữ liệu website và kiểm tra: \texttt{python3 scripts/build\_data.py} rồi \texttt{python3 scripts/validate\_data.py}
  \item Xem website trên máy: \texttt{python3 -m http.server 8000}, mở địa chỉ \texttt{http://localhost:8000}
\end{enumerate}
